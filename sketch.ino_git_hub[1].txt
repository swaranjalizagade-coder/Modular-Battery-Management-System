#define BLYNK_TEMPLATE_ID "your_templet_id"
#define BLYNK_TEMPLATE_NAME "BMS Monitoring System"
#define BLYNK_AUTH_TOKEN "your_private_token"

#define BLYNK_PRINT Serial

#include <WiFi.h>
#include <BlynkSimpleEsp32.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ---------------- WIFI ----------------

char ssid[] = "Wokwi-GUEST";
char pass[] = "";

// ---------------- OLED ----------------

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

Adafruit_SSD1306 display(
  SCREEN_WIDTH,
  SCREEN_HEIGHT,
  &Wire,
  OLED_RESET
);

// ---------------- PINS ----------------

#define CELL_COUNT 4

const int CELL_PINS[CELL_COUNT] = {
  32, 33, 34, 35
};

#define ACK_BUTTON 25
#define WARNING_BUTTON 26
#define CRITICAL_BUTTON 27

#define RELAY_PIN 23

#define GREEN_LED 16
#define YELLOW_LED 17
#define RED_LED 18

// ---------------- TIMERS ----------------

BlynkTimer timer;

unsigned long lastConnectionAttempt = 0;

// ---------------- BATTERY DATA ----------------

float cellVoltage[CELL_COUNT];

float weakestCell = 0;
float strongestCell = 0;

float batterySoC = 0;
float cellImbalance = 0;
float imbalanceTrend = 0;

float previousImbalance = 0;

int riskScore = 0;
float batteryHealth = 100;

int totalFaultCount = 0;
int severityLevel = 0;

int wifiRSSI = 0;
int queueDepth = 0;

bool manualWarning = false;
bool manualCritical = false;

bool previousWarningButton = HIGH;
bool previousCriticalButton = HIGH;
bool previousAckButton = HIGH;

bool faultActive = false;

String operatingState = "NORMAL";

String recommendation =
  "Battery operating normally";

// ---------------- THRESHOLDS ----------------

const float MIN_CELL_VOLTAGE = 2.7;
const float MAX_CELL_VOLTAGE = 4.2;

const float WARNING_IMBALANCE = 0.15;
const float CRITICAL_IMBALANCE = 0.30;

const float LOW_CELL_WARNING = 3.0;
const float CRITICAL_CELL = 2.8;

// ---------------- READ BATTERY ----------------

void readBattery() {

  weakestCell = 99;
  strongestCell = 0;

  float totalVoltage = 0;

  for (int i = 0; i < CELL_COUNT; i++) {

    int rawValue = analogRead(CELL_PINS[i]);

    cellVoltage[i] =
      MIN_CELL_VOLTAGE +
      (rawValue / 4095.0) *
      (MAX_CELL_VOLTAGE - MIN_CELL_VOLTAGE);

    totalVoltage += cellVoltage[i];

    if (cellVoltage[i] < weakestCell) {
      weakestCell = cellVoltage[i];
    }

    if (cellVoltage[i] > strongestCell) {
      strongestCell = cellVoltage[i];
    }
  }

  cellImbalance = strongestCell - weakestCell;

  batterySoC =
    ((totalVoltage / CELL_COUNT - MIN_CELL_VOLTAGE) /
    (MAX_CELL_VOLTAGE - MIN_CELL_VOLTAGE)) * 100.0;

  batterySoC = constrain(batterySoC, 0, 100);

  imbalanceTrend =
    cellImbalance - previousImbalance;

  previousImbalance = cellImbalance;
}

// ---------------- READ BUTTONS ----------------

void readButtons() {

  bool warningButton = digitalRead(WARNING_BUTTON);
  bool criticalButton = digitalRead(CRITICAL_BUTTON);
  bool ackButton = digitalRead(ACK_BUTTON);

  // Yellow button: warning

  if (previousWarningButton == HIGH &&
      warningButton == LOW) {

    manualWarning = true;
  }

  // Red button: critical

  if (previousCriticalButton == HIGH &&
      criticalButton == LOW) {

    manualCritical = true;
  }

  // Blue button: acknowledge

  if (previousAckButton == HIGH &&
      ackButton == LOW) {

    manualWarning = false;
    manualCritical = false;
  }

  previousWarningButton = warningButton;
  previousCriticalButton = criticalButton;
  previousAckButton = ackButton;
}

// ---------------- CALCULATE RISK ----------------

void calculateRisk() {

  float imbalanceRisk =
    constrain((cellImbalance / 0.30) * 100.0, 0, 100);

  float faultRisk =
    constrain(totalFaultCount * 10.0, 0, 100);

  float lowSoCRisk =
    100.0 - batterySoC;

  float trendRisk =
    constrain(abs(imbalanceTrend) * 200.0, 0, 100);

  riskScore =
    (int)(
      imbalanceRisk * 0.35 +
      faultRisk * 0.25 +
      lowSoCRisk * 0.25 +
      trendRisk * 0.15
    );

  riskScore = constrain(riskScore, 0, 100);

  batteryHealth = 100.0 - riskScore;
}

// ---------------- SYSTEM STATE ----------------

void updateSystemState() {

  String newState = "NORMAL";

  bool criticalCondition = false;
  bool warningCondition = false;

  for (int i = 0; i < CELL_COUNT; i++) {

    if (cellVoltage[i] <= CRITICAL_CELL) {
      criticalCondition = true;
    }

    if (cellVoltage[i] <= LOW_CELL_WARNING) {
      warningCondition = true;
    }
  }

  if (cellImbalance >= CRITICAL_IMBALANCE ||
      criticalCondition ||
      manualCritical) {

    newState = "FAILSAFE";
  }

  else if (cellImbalance >= WARNING_IMBALANCE ||
           warningCondition ||
           manualWarning) {

    newState = "DEGRADED";
  }

  if (newState != operatingState) {

    if (newState != "NORMAL") {
      totalFaultCount++;
    }

    operatingState = newState;
  }

  if (operatingState == "NORMAL") {

    severityLevel = 0;

    recommendation =
      "Battery operating normally";
  }

  else if (operatingState == "DEGRADED") {

    severityLevel = 1;

    recommendation =
      "Check cell balance and inspect battery conditions";
  }

  else {

    severityLevel = 2;

    recommendation =
      "Critical condition: inspect battery immediately";
  }

  faultActive = (operatingState != "NORMAL");
}

// ---------------- OUTPUT CONTROL ----------------

void updateOutputs() {

  if (operatingState == "NORMAL") {

    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(YELLOW_LED, LOW);
    digitalWrite(RED_LED, LOW);

    // Relay ON

    digitalWrite(RELAY_PIN, HIGH);
  }

  else if (operatingState == "DEGRADED") {

    digitalWrite(GREEN_LED, LOW);
    digitalWrite(YELLOW_LED, HIGH);
    digitalWrite(RED_LED, LOW);

    // Relay ON

    digitalWrite(RELAY_PIN, HIGH);
  }

  else {

    digitalWrite(GREEN_LED, LOW);
    digitalWrite(YELLOW_LED, LOW);
    digitalWrite(RED_LED, HIGH);

    // Relay OFF

    digitalWrite(RELAY_PIN, LOW);
  }
}

// ---------------- OLED DISPLAY ----------------

void updateOLED() {

  display.clearDisplay();

  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);

  display.setCursor(0, 0);
  display.println("BMS TASK 6");

  display.setCursor(0, 12);
  display.print("State: ");
  display.println(operatingState);

  display.setCursor(0, 22);
  display.print("SoC: ");
  display.print(batterySoC, 1);
  display.println("%");

  display.setCursor(0, 32);
  display.print("Imbalance: ");
  display.print(cellImbalance, 2);
  display.println("V");

  display.setCursor(0, 42);
  display.print("Risk: ");
  display.print(riskScore);
  display.print("/100");

  display.setCursor(0, 52);
  display.print("Faults: ");
  display.println(totalFaultCount);

  display.display();
}

// ---------------- SERIAL MONITOR ----------------

void printSerialData() {

  Serial.println("--------------------------");

  Serial.print("Wi-Fi: ");
  Serial.println(WiFi.status() == WL_CONNECTED ?
                 "CONNECTED" : "DISCONNECTED");

  Serial.print("Blynk: ");
  Serial.println(Blynk.connected() ?
                 "CONNECTED" : "DISCONNECTED");

  for (int i = 0; i < CELL_COUNT; i++) {

    Serial.print("Cell ");
    Serial.print(i + 1);
    Serial.print(": ");
    Serial.print(cellVoltage[i], 2);
    Serial.println(" V");
  }

  Serial.print("SoC: ");
  Serial.println(batterySoC);

  Serial.print("Imbalance: ");
  Serial.println(cellImbalance);

  Serial.print("Risk score: ");
  Serial.println(riskScore);

  Serial.print("Battery health: ");
  Serial.println(batteryHealth);

  Serial.print("Status: ");
  Serial.println(operatingState);

  Serial.print("Fault count: ");
  Serial.println(totalFaultCount);

  Serial.print("Recommendation: ");
  Serial.println(recommendation);

  Serial.println("--------------------------");
}

// ---------------- BLYNK PUBLISHING ----------------

void publishBlynkData() {

  if (!Blynk.connected()) {
    return;
  }

  // TASK 5 DATA STREAMS

  Blynk.virtualWrite(V0, cellVoltage[0]);
  Blynk.virtualWrite(V1, cellVoltage[1]);
  Blynk.virtualWrite(V2, cellVoltage[2]);
  Blynk.virtualWrite(V3, cellVoltage[3]);

  Blynk.virtualWrite(V4, weakestCell);
  Blynk.virtualWrite(V5, strongestCell);

  Blynk.virtualWrite(V6,
    operatingState == "FAILSAFE" ? 0 : 1);

  Blynk.virtualWrite(V7,
    faultActive ? "FAULT" : "NORMAL");

  Blynk.virtualWrite(V8, operatingState);

  Blynk.virtualWrite(V9,
    WiFi.RSSI());

  Blynk.virtualWrite(V10, queueDepth);

  Blynk.virtualWrite(V11, "WOKWI");

  // TASK 6 DATA STREAMS

  Blynk.virtualWrite(V12, batterySoC);

  Blynk.virtualWrite(V13, cellImbalance);

  Blynk.virtualWrite(V14, imbalanceTrend);

  Blynk.virtualWrite(V15, riskScore);

  Blynk.virtualWrite(V16, batteryHealth);

  Blynk.virtualWrite(V17, millis() / 1000);

  Blynk.virtualWrite(V18, totalFaultCount);

  Blynk.virtualWrite(V19, severityLevel);

  Blynk.virtualWrite(V20, recommendation);

  Serial.println("Task 5 and Task 6 data sent to Blynk");
}

// ---------------- MAIN UPDATE ----------------

void updateBMS() {

  readBattery();

  readButtons();

  calculateRisk();

  updateSystemState();

  updateOutputs();

  updateOLED();

  printSerialData();
}

// ---------------- BLYNK CONNECTION ----------------

void reconnectBlynk() {

  if (WiFi.status() != WL_CONNECTED) {
    return;
  }

  if (!Blynk.connected()) {

    Serial.println("Connecting to Blynk Cloud...");

    if (Blynk.connect(1000)) {
      Serial.println("Blynk Connected Successfully!");
    }

    else {
      Serial.println("Blynk Connection Failed!");
    }
  }
}

// ---------------- SETUP ----------------

void setup() {

  Serial.begin(115200);

  // Configure pins

  for (int i = 0; i < CELL_COUNT; i++) {
    pinMode(CELL_PINS[i], INPUT);
  }

  pinMode(ACK_BUTTON, INPUT_PULLUP);
  pinMode(WARNING_BUTTON, INPUT_PULLUP);
  pinMode(CRITICAL_BUTTON, INPUT_PULLUP);

  pinMode(RELAY_PIN, OUTPUT);

  pinMode(GREEN_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);

  // Initial outputs

  digitalWrite(RELAY_PIN, HIGH);

  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);
  digitalWrite(RED_LED, LOW);

  // Initialize OLED

  Wire.begin(21, 22);

  if (!display.begin(
        SSD1306_SWITCHCAPVCC,
        0x3C)) {

    Serial.println("OLED initialization failed!");
  }

  else {

    display.clearDisplay();
    display.setTextSize(1);
    display.setTextColor(SSD1306_WHITE);
    display.setCursor(0, 0);
    display.println("BMS TASK 6");
    display.println("Starting system...");
    display.display();
  }

  // Connect Wi-Fi

  WiFi.begin(ssid, pass);

  Serial.println("Connecting to Wi-Fi...");

  unsigned long startTime = millis();

  while (WiFi.status() != WL_CONNECTED &&
         millis() - startTime < 15000) {

    delay(500);
    Serial.print(".");
  }

  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("Wi-Fi Connected!");
  }

  else {
    Serial.println("Wi-Fi Connection Failed!");
  }

  // Configure Blynk using existing device

  Blynk.config(BLYNK_AUTH_TOKEN);

  if (WiFi.status() == WL_CONNECTED) {

    Serial.println("Connecting to Blynk...");

    if (Blynk.connect(10000)) {
      Serial.println("Blynk Connected Successfully!");
    }

    else {
      Serial.println("Blynk Connection Failed!");
    }
  }

  // Timers

  timer.setInterval(1000L, updateBMS);

  timer.setInterval(2000L, publishBlynkData);

  timer.setInterval(5000L, reconnectBlynk);
}

// ---------------- LOOP ----------------

void loop() {

  Blynk.run();

  timer.run();
}
